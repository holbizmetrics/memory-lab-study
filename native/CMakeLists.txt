cmake_minimum_required(VERSION 3.22)
project(MemoryLabNative VERSION 0.1.0 LANGUAGES CXX)
add_executable(memorylab_native main.cpp)
target_compile_features(memorylab_native PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(memorylab_native PRIVATE /W4 /fp:precise)
else()
 target_compile_options(memorylab_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME native_core COMMAND memorylab_native --test)
add_test(NAME arpeggiator COMMAND memorylab_native --arp-test)

add_executable(memorylab_voice_drive voice-drive-study.cpp)
target_compile_features(memorylab_voice_drive PRIVATE cxx_std_17)
target_compile_definitions(memorylab_voice_drive PRIVATE MEMORYLAB_EXPERIMENTAL_VOICE_DRIVE=1)
if(MSVC)
 target_compile_options(memorylab_voice_drive PRIVATE /W4 /fp:precise)
endif()
